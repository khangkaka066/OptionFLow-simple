//====================================================

f_delete_line(line _line) =>
    if not na(_line)
        line.delete(_line)

f_delete_label(label _label) =>
    if not na(_label)
        label.delete(_label)

f_delete_box(box _box) =>
    if not na(_box)
        box.delete(_box)

f_delete_line_array(array<line> _lines) =>
    if array.size(_lines) > 0
        for i = 0 to array.size(_lines) - 1
            item = array.get(_lines, i)
            if not na(item)
                line.delete(item)
    array.clear(_lines)

f_delete_label_array(array<label> _labels) =>
    if array.size(_labels) > 0
        for i = 0 to array.size(_labels) - 1
            item = array.get(_labels, i)
            if not na(item)
                label.delete(item)
    array.clear(_labels)

f_delete_box_array(array<box> _boxes) =>
    if array.size(_boxes) > 0
        for i = 0 to array.size(_boxes) - 1
            item = array.get(_boxes, i)
            if not na(item)
                box.delete(item)
    array.clear(_boxes)

f_update_level_box(box _box, bool _show, float _price, color _color) =>
    box result = _box
    if _show and customBoxSize > 0
        halfSize = customBoxSize * 0.5
        topPrice = _price + halfSize
        bottomPrice = _price - halfSize
        fillColor = color.new(_color, customBoxTransparency)
        if na(result)
            result := box.new(bar_index - 1, topPrice, bar_index + 60, bottomPrice, xloc=xloc.bar_index, bgcolor=fillColor, border_color=_color, border_width=1, extend=extend.left)
        else
            box.set_lefttop(result, bar_index - 1, topPrice)
            box.set_rightbottom(result, bar_index + 60, bottomPrice)
            box.set_bgcolor(result, fillColor)
            box.set_border_color(result, _color)
            box.set_extend(result, extend.left)
    else if not na(result)
        box.delete(result)
        result := na
    result

f_eq(float a, float b) =>
    not na(a) and not na(b) and math.abs(a - b) <= syminfo.mintick * mergeTicks

f_add(string txt, string item) =>
    txt == "" ? item : txt + " / " + item

customLabelX() =>
    bar_index

// Custom levels stop 100 bars to the right and extend only to the left.
customViewportWidth() =>
    math.max(chart.right_visible_bar_time - chart.left_visible_bar_time, 1)

customPriceScaleLabelBar() =>
    int barMilliseconds = timeframe.in_seconds() * 1000
    int visibleBars = int(math.round(customViewportWidth() / barMilliseconds))
    bar_index + math.min(500, math.max(25, int(math.round(visibleBars * 0.80))))

f_draw_pre_level(int _idx, float _price, string _text, color _color, bool _showLabel, float _labelPrice) =>
    while array.size(preMarketLines) <= _idx
        array.push(preMarketLines, na)
    while array.size(preMarketLabels) <= _idx
        array.push(preMarketLabels, na)
    rightX = na(preMarketEndTime) ? bar_index : preMarketEndTime
    itemLine = array.get(preMarketLines, _idx)
    if na(itemLine)
        itemLine := line.new(preMarketStartTime, _price, rightX, _price, xloc=xloc.bar_index, color=_color, width=preMarketWidth, extend=extend.none)
        array.set(preMarketLines, _idx, itemLine)
    else
        line.set_xy1(itemLine, preMarketStartTime, _price)
        line.set_xy2(itemLine, rightX, _price)
        line.set_color(itemLine, _color)
        line.set_width(itemLine, preMarketWidth)
    itemLabel = array.get(preMarketLabels, _idx)
    if _showLabel
        if na(itemLabel)
            itemLabel := label.new(rightX, _labelPrice, _text, xloc=xloc.bar_index, style=label.style_label_left, color=#00000000, textcolor=_color, size=labelSize)
            array.set(preMarketLabels, _idx, itemLabel)
        else
            label.set_xy(itemLabel, rightX, _labelPrice)
            label.set_text(itemLabel, _text)
            label.set_textcolor(itemLabel, _color)
            label.set_size(itemLabel, labelSize)
    else if not na(itemLabel)
        label.delete(itemLabel)
        array.set(preMarketLabels, _idx, na)

f_draw_pre_box(int _idx, float _top, float _bottom, color _color) =>
    while array.size(preMarketBoxes) <= _idx
        array.push(preMarketBoxes, na)
    rightX = na(preMarketEndTime) ? bar_index : preMarketEndTime
    topPrice = math.max(_top, _bottom)
    bottomPrice = math.min(_top, _bottom)
    itemBox = array.get(preMarketBoxes, _idx)
    if na(itemBox)
        itemBox := box.new(preMarketStartTime, topPrice, rightX, bottomPrice, xloc=xloc.bar_index, bgcolor=color.new(_color, customBoxTransparency), border_color=_color, border_width=1, extend=extend.none)
        array.set(preMarketBoxes, _idx, itemBox)
    else
        box.set_lefttop(itemBox, preMarketStartTime, topPrice)
        box.set_rightbottom(itemBox, rightX, bottomPrice)
        box.set_bgcolor(itemBox, color.new(_color, customBoxTransparency))
        box.set_border_color(itemBox, _color)
        box.set_extend(itemBox, extend.none)
